// ===================================================== // DOM REFERENCES // ===================================================== const scanBtn = document.getElementById("scanBtn"); const video = document.getElementById("video"); const canvas = document.getElementById("canvas"); const statusText = document.getElementById("status"); let activeStream = null; let isProcessing = false; // ===================================================== // HELPERS // ===================================================== function setStatus(message, type = "") { statusText.textContent = message; statusText.className = `status ${type}`; } function wait(ms) { return new Promise(resolve => setTimeout(resolve, ms)); } // ===================================================== // CAMERA START // ===================================================== async function startCamera(mode = "environment") { try { // Extra cleanup before switching cameras await stopCamera(); const constraints = { audio: false, video: { facingMode: { ideal: mode }, width: { ideal: 1280 }, height: { ideal: 720 } } }; const stream = await navigator.mediaDevices.getUserMedia(constraints); activeStream = stream; video.srcObject = stream; video.style.display = "block"; await video.play(); // Mobile hardware stabilization await wait(600); return stream; } catch (error) { console.error("Camera start error:", error); throw error; } } // ===================================================== // CAPTURE FRAME // ===================================================== function captureFrame() { const ctx = canvas.getContext("2d"); canvas.width = video.videoWidth || 1280; canvas.height = video.videoHeight || 720; ctx.drawImage(video, 0, 0, canvas.width, canvas.height); return canvas.toDataURL("image/jpeg", 0.82); } // ===================================================== // AGGRESSIVE CAMERA CLEANUP // ===================================================== async function stopCamera() { try { if (activeStream) { activeStream.getTracks().forEach(track => { try { track.stop(); } catch (e) { console.warn(e); } }); } video.pause(); video.srcObject = null; video.removeAttribute("src"); video.load(); activeStream = null; // Give browser time to release hardware await wait(700); } catch (error) { console.error("Camera cleanup error:", error); } } // ===================================================== // DATABASE SAVE // ===================================================== async function saveToFirestore(rearSnapshot, frontSnapshot) { try { await addDoc(collection(db, "logs"), { timestamp: serverTimestamp(), rearSnapshot, frontSnapshot, action: "Guest Entry" }); } catch (error) { console.error("Firestore Error:", error); throw error; } } // ===================================================== // SAFE NOTIFICATION // ===================================================== function triggerNotification() { // FULLY DECOUPLED FROM FIREBASE + CAMERA THREAD setTimeout(async () => { try { if (!("Notification" in window)) return; if (Notification.permission === "default") { await Notification.requestPermission(); } if (Notification.permission === "granted") { new Notification("Gate Alert!", { body: "Guest captured successfully.", icon: "https://cdn-icons-png.flaticon.com/512/6195/6195700.png", silent: false }); } } catch (error) { console.error("Notification Error:", error); } }, 900); } // ===================================================== // MAIN EXECUTION FLOW // ===================================================== async function handleGuestScan() { if (isProcessing) return; isProcessing = true; scanBtn.disabled = true; let rearSnapshot = null; let frontSnapshot = null; try { // ================================================= // STEP 1 -> REAR CAMERA // ================================================= setStatus("Opening rear camera..."); await startCamera("environment"); setStatus("Scanning gate / QR..."); await wait(2000); rearSnapshot = captureFrame(); // ================================================= // STEP 2 -> STOP REAR CAMERA // ================================================= setStatus("Closing rear camera..."); await stopCamera(); // ================================================= // STEP 3 -> FRONT CAMERA // ================================================= setStatus("Opening front camera..."); await startCamera("user"); setStatus("Capturing guest face..."); await wait(1800); frontSnapshot = captureFrame(); // ================================================= // STEP 4 -> STOP FRONT CAMERA // ================================================= setStatus("Releasing camera hardware..."); await stopCamera(); // ================================================= // STEP 5 -> FIRE NOTIFICATION IMMEDIATELY // ================================================= triggerNotification(); // ================================================= // STEP 6 -> DATABASE WRITE // ================================================= setStatus("Uploading to database..."); await saveToFirestore(rearSnapshot, frontSnapshot); // ================================================= // COMPLETE // ================================================= setStatus("Guest entry completed successfully.", "success"); } catch (error) { console.error(error); await stopCamera(); setStatus( "Error: " + (error.message || "Execution failed."), "error" ); } finally { isProcessing = false; scanBtn.disabled = false; } } // ===================================================== // BUTTON EVENT // ===================================================== scanBtn.addEventListener("click", handleGuestScan);
